import { z } from "zod";

// ─── Message Schema ────────────────────────────────────────────────────────────
export const MessageSchema = z.object({
  role: z.enum(["system", "user", "assistant", "function"]),
  content: z.string().min(1, "Message content cannot be empty"),
  name: z.string().optional(), // Used for function messages
});

// ─── Engine Config Schema ──────────────────────────────────────────────────────
export const EngineConfigSchema = z.object({
  model: z.string().default("gpt-4o"),
  maxTokens: z.number().int().positive().max(128000).default(4096),
  temperature: z.number().min(0).max(2).default(0.7),
  topP: z.number().min(0).max(1).default(1),
  frequencyPenalty: z.number().min(-2).max(2).default(0),
  presencePenalty: z.number().min(-2).max(2).default(0),
  systemPrompt: z.string().optional(),
  stream: z.boolean().default(false),
  maxRetries: z.number().int().positive().default(3),
  timeout: z.number().int().positive().default(30000),
});

// ─── Response Schema ───────────────────────────────────────────────────────────
export const EngineResponseSchema = z.object({
  id: z.string(),
  content: z.string(),
  model: z.string(),
  usage: z.object({
    promptTokens: z.number(),
    completionTokens: z.number(),
    totalTokens: z.number(),
  }),
  finishReason: z.enum(["stop", "length", "content_filter", "tool_calls", "null"]),
  latencyMs: z.number(),
  timestamp: z.date(),
});

// ─── Exported Types ────────────────────────────────────────────────────────────
export type Message          = z.infer<typeof MessageSchema>;
export type EngineConfig     = z.infer<typeof EngineConfigSchema>;
export type EngineResponse   = z.infer<typeof EngineResponseSchema>;

// ─── Custom Error Types ────────────────────────────────────────────────────────
export class NeuralEngineError extends Error {
  constructor(
    message: string,
    public readonly code: string,
    public readonly statusCode?: number,
    public readonly retryable: boolean = false
  ) {
    super(message);
    this.name = "NeuralEngineError";
    Object.setPrototypeOf(this, NeuralEngineError.prototype);
  }
}

export class RateLimitError extends NeuralEngineError {
  constructor(public readonly retryAfter?: number) {
    super("Rate limit exceeded", "RATE_LIMIT", 429, true);
    this.name = "RateLimitError";
  }
}

export class TokenLimitError extends NeuralEngineError {
  constructor(public readonly tokenCount: number, public readonly limit: number) {
    super(
      `Token limit exceeded: ${tokenCount} tokens (limit: ${limit})`,
      "TOKEN_LIMIT",
      400,
      false
    );
    this.name = "TokenLimitError";
  }
}

export class AuthenticationError extends NeuralEngineError {
  constructor() {
    super("Invalid or missing API key", "AUTH_ERROR", 401, false);
    this.name = "AuthenticationError";
  }
}
